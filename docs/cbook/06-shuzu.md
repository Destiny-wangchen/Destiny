# 第 6 章 · 数组

数组把「一堆同类型的数据」打包存进连续内存，从此数据有了编号：`a[i]` 一写，循环就能批量处理。本章的题目几乎覆盖了数组操作的完整工具箱——计数标记、前缀累计、查找、排序、二维网格。

必须养成的基本功：

- **下标从 0 还是 1**：算法描述常用 0 起，生活语言常用 1 起，读写时先统一口径，输出时再换算。
- **数组要开够**：n ≤ 10⁶ 就开 static 大数组或用堆；数组名本身是指针，越界不报错但会埋雷。
- **查找用标记数组**：判断「某数出现过没有」「哪个编号没来」，开一个 seen[] 打标记，比嵌套循环快一个量级。

排序题虽然「什么排序都可以」，但插入排序十几行就能写对，是理解排序原理的最佳起点：把新元素往前挪，插到该在的位置。

```c
for (i = 1; i < n; ++i) {
    t = a[i];
    for (j = i - 1; j >= 0 && a[j] > t; --j) a[j + 1] = a[j];
    a[j + 1] = t;
}
```

**二维数组**（扫雷、导航、模拟游戏）本质是「数组套数组」，双层下标 (r, c) 对应棋盘思维；处理边界用 `x >= 0 && x < n && y >= 0 && y < m` 的卫语句，把越界挡在取值之前。

本章共 7 道题：2 道精讲例题 + 5 道自选练习。

## 精讲例题

### 例题 1 · 给数据排个队

输入元素个数和一组数据，升序排序后输出，数字之间一个空格，最后一个数后面没有空格。

**输入**：第一行元素个数（≤100），第二行数据。  
**输出**：有序序列一行。

**样例**

输入：
```text
6
10 90 80 30 20 15
```
输出：
```text
10 15 20 30 80 90
```

**样例**

输入：
```text
5
10 10 10 10 10
```
输出：
```text
10 10 10 10 10
```

**思路**：插入排序三步走——拿出待插元素、把比它大的元素整体右移一格、落位。输出时注意「最后一个数后面没有空格」：用三目运算符控制分隔符，最后一项输出换行、其余输出空格。

多写几轮手工模拟，比背代码更能记住插入排序为什么对。

```c
/* 52【应用】排序：升序输出，数字间一个空格，末尾无空格 */
#include <stdio.h>

int main(void)
{
    int n, a[105], i, j, t;

    if (scanf("%d", &n) != 1) return 0;
    for (i = 0; i < n; ++i)
        if (scanf("%d", &a[i]) != 1) return 0;

    for (i = 1; i < n; ++i) {
        t = a[i];
        for (j = i - 1; j >= 0 && a[j] > t; --j) a[j + 1] = a[j];
        a[j + 1] = t;
    }
    for (i = 0; i < n; ++i)
        printf("%d%c", a[i], i == n - 1 ? '\n' : ' ');
    return 0;
}
```

### 例题 2 · 扫雷计数器

给定若干张 n×m 雷区图（`*` 是雷，`.` 是空地），对每个空地格子输出周围 8 格中雷的数量，雷的位置原样保留 `*`。每张图前输出 `Field #x:`（x 从 1 起），相邻两张图之间空一行；读到 `0 0` 结束。

**输入**：多张地图，每张先给 n m，随后 n 行字符串。  
**输出**：按上述格式输出所有图。

**样例**

输入：
```text
4 4
*...
....
.*..
....
3 5
**...
.....
.*...
0 0
```
输出：
```text
Field #1:
*100
2210
1*10
1110

Field #2:
**100
33200
1*100
```

**样例**

输入：
```text
3 3
...
...
...
3 3
***
***
***
0 0
```
输出：
```text
Field #1:
000
000
000

Field #2:
***
***
***
```

**思路**：扫雷只有一件事——对每个非雷格子数周围 8 格里有几颗雷。把 8 个邻居写成双层偏移循环（di、dj 各取 -1..1），代码从 8 个 if 缩成 4 行。

两个工程细节：多组数据直到 0 0；两个地图之间输出一个空行。边界格子的邻居会出界，全靠卫语句拦住。

```c
/* 45【游戏】扫雷：给每个非雷格子统计周围 8 格的地雷数 */
#include <stdio.h>

static char g[105][105];

int main(void)
{
    int n, m, field = 0, i, j, di, dj;

    while (scanf("%d %d", &n, &m) == 2) {
        if (n == 0 && m == 0) break;
        for (i = 0; i < n; ++i)
            if (scanf("%104s", g[i]) != 1) return 0;

        if (field > 0) putchar('\n');
        printf("Field #%d:\n", ++field);

        for (i = 0; i < n; ++i) {
            for (j = 0; j < m; ++j) {
                if (g[i][j] == '*') { putchar('*'); continue; }
                int cnt = 0;
                for (di = -1; di <= 1; ++di)
                    for (dj = -1; dj <= 1; ++dj) {
                        int x = i + di, y = j + dj;
                        if ((di || dj) && x >= 0 && x < n && y >= 0 && y < m && g[x][y] == '*')
                            ++cnt;
                    }
                putchar('0' + cnt);
            }
            putchar('\n');
        }
    }
    return 0;
}
```

```c
/* 50【学长出题】参观军事博物馆：逐站计算拥挤程度并找出最舒服的站 */
#include <stdio.h>

int main(void)
{
    static int pax[10005];        /* 各站关门后车上人数 */
    static int minIdx[10005];
    int n, m, k, i, minCnt = 0, best;

    if (scanf("%d %d", &n, &m) != 2) return 0;
    if (scanf("%d", &k) != 1) return 0;
    for (i = 1; i <= n - 1; ++i) {
        int a, b;
        if (scanf("%d %d", &a, &b) != 2) return 0;
        k = k + a - b;
        pax[i] = k;
        printf("%.3f\n", (double)k / m);
    }
    best = pax[1];
    for (i = 2; i <= n - 1; ++i)
        if (pax[i] < best) best = pax[i];
    for (i = 1; i <= n - 1; ++i)
        if (pax[i] == best) minIdx[minCnt++] = i;
    for (i = 0; i < minCnt; ++i)
        printf("%d,", minIdx[i]);
    putchar('\n');
    return 0;
}
```

```c
/* 51【应用】参观抗日战争纪念馆：找出未签到的同学编号 */
#include <stdio.h>

int main(void)
{
    static int arrived[1005];
    int n, m, i, x, missing = 0;

    if (scanf("%d %d", &n, &m) != 2) return 0;
    for (i = 0; i < m; ++i) {
        if (scanf("%d", &x) != 1) return 0;
        if (x >= 0 && x < n) arrived[x] = 1;
    }
    for (i = 0; i < n; ++i)
        if (!arrived[i]) ++missing;
    if (missing == 0) {
        printf("Go!\n");
        return 0;
    }
    for (i = 0; i < n; ++i)
        if (!arrived[i]) printf("%d%c", i, (--missing) ? ' ' : '\n');
    return 0;
}
```

::: details 练习 4 · 把队伍排齐最少走几步（参考代码）
```c
/* H18【学长出题】北理工的勇士：排序后取中位数，纵坐标取 y 的中位数，横坐标取 x-i 的中位数 */
#include <stdio.h>
#include <stdlib.h>

static int xs[10005], ys[10005];

static int cmp_int(const void *p, const void *q)
{
    int a = *(const int *)p, b = *(const int *)q;
    return (a > b) - (a < b);
}

int main(void)
{
    int n, i;
    long long cost = 0;
    int medX, medY;

    if (scanf("%d", &n) != 1) return 0;
    for (i = 0; i < n; ++i)
        if (scanf("%d %d", &xs[i], &ys[i]) != 2) return 0;

    qsort(xs, (size_t)n, sizeof(int), cmp_int);
    qsort(ys, (size_t)n, sizeof(int), cmp_int);

    /* 横坐标：把第 i 个勇士对齐到 X+i，等价于让 x[i]-i 尽量接近同一个 X */
    for (i = 0; i < n; ++i) xs[i] -= i;
    qsort(xs, (size_t)n, sizeof(int), cmp_int);

    medX = xs[n / 2];
    medY = ys[n / 2];
    for (i = 0; i < n; ++i) {
        cost += llabs((long long)xs[i] - medX);
        cost += llabs((long long)ys[i] - medY);
    }
    printf("%lld\n", cost);
    return 0;
}
```

::: details 练习 5 · 音游谱面分析（参考代码）
```c
/* H19【学长出题】音乐游戏：统计 tap（孤立音符）与 hold（连续 >=2 行的音符）
 * 读入到"连续三行没有任何音符"为止，逐列扫描竖向连续段 */
#include <stdio.h>
#include <string.h>

int main(void)
{
    static int run[64];        /* 每列当前连续音符长度 */
    int cols, i, col;
    char line[256];
    int emptyRun = 0;
    long long tap = 0, hold = 0;

    if (scanf("%d", &cols) != 1) return 0;
    fgets(line, sizeof(line), stdin);            /* 吃掉行尾 */

    while (fgets(line, sizeof(line), stdin)) {
        int hasNote = 0;

        line[strcspn(line, "\r\n")] = '\0';
        if ((int)strlen(line) < cols) {
            /* 行太短则补 '-'，避免越界 */
            int l = (int)strlen(line);
            for (i = l; i < cols; ++i) line[i] = '-';
            line[cols] = '\0';
        }

        for (col = 0; col < cols; ++col) {
            if (line[col] == '#') {
                ++run[col];
                hasNote = 1;
            } else if (run[col] > 0) {
                if (run[col] == 1) ++tap;
                else ++hold;
                run[col] = 0;
            }
        }

        if (hasNote) emptyRun = 0;
        else if (++emptyRun >= 3) break;         /* 连续三行无音符：谱面结束 */
    }

    /* 收尾：谱面在最后一行直接结束时把未结算的段补上 */
    for (col = 0; col < cols; ++col) {
        if (run[col] == 1) ++tap;
        else if (run[col] >= 2) ++hold;
    }

    printf("%lld\n%lld\n", tap, hold);
    return 0;
}
```
